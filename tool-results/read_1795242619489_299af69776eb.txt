     1→"use client";
     2→
     3→/**
     4→ * المخزون والمستودعات — حركات، استلام، تحويل، تسوية، جرد، مراقبة صلاحية
     5→ */
     6→import { useCallback, useEffect, useMemo, useState } from "react";
     7→import { useApp } from "@/lib/store";
     8→import { useI18n, PageHeader, EmptyState, Pagination } from "./shared";
     9→import { api, ApiError } from "@/lib/client-api";
    10→import { fmtMoney, fmtDate, fmtNum, expiryStatus } from "@/lib/format";
    11→import { Button } from "@/components/ui/button";
    12→import { Input } from "@/components/ui/input";
    13→import { Label } from "@/components/ui/label";
    14→import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
    15→import { Badge } from "@/components/ui/badge";
    16→import { Tabs, TabsContent, TabsList, TabsTrigger } from "@/components/ui/tabs";
    17→import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from "@/components/ui/select";
    18→import { Table, TableBody, TableCell, TableHead, TableHeader, TableRow } from "@/components/ui/table";
    19→import { Dialog, DialogContent, DialogHeader, DialogTitle } from "@/components/ui/dialog";
    20→import { ScrollArea } from "@/components/ui/scroll-area";
    21→import { toast } from "sonner";
    22→import {
    23→  Warehouse, ArrowLeftRight, PackagePlus, Scale, ClipboardCheck,
    24→  CalendarClock, Search, RefreshCw, Plus, Trash2,
    25→} from "lucide-react";
    26→import { cn } from "@/lib/utils";
    27→
    28→export default function InventoryScreen() {
    29→  const { t, lang } = useI18n();
    30→  const { branches } = useApp();
    31→  const [tab, setTab] = useState("movements");
    32→
    33→  const fmtM = (n: number) => fmtMoney(n, lang);
    34→  const fmtQ = (n: number) => fmtNum(n, lang);
    35→
    36→  // ─── الحركات ───
    37→  const [movements, setMovements] = useState<any[]>([]);
    38→  const [movTotal, setMovTotal] = useState(0);
    39→  const [movPage, setMovPage] = useState(1);
    40→  const [movType, setMovType] = useState("all");
    41→  const [movBranch, setMovBranch] = useState("all");
    42→
    43→  const loadMovements = useCallback(async () => {
    44→    try {
    45→      const params = new URLSearchParams({ page: String(movPage), limit: "15", view: "movements" });
    46→      if (movType !== "all") params.set("type", movType);
    47→      if (movBranch !== "all") params.set("branchId", movBranch);
    48→      const data = await api(`/stock?${params}`);
    49→      setMovements(data.items || []);
    50→      setMovTotal(data.total || 0);
    51→    } catch (e) {
    52→      toast.error(e instanceof Error ? e.message : t("app.error"));
    53→    }
    54→     
    55→  }, [movPage, movType, movBranch]);
    56→
    57→  // ─── التشغيلات والصلاحية ───
    58→  const [batches, setBatches] = useState<any[]>([]);
    59→  const [batchTotal, setBatchTotal] = useState(0);
    60→  const [batchPage, setBatchPage] = useState(1);
    61→  const [batchFilter, setBatchFilter] = useState("all");
    62→  const [batchBranch, setBatchBranch] = useState("all");
    63→
    64→  const loadBatches = useCallback(async () => {
    65→    try {
    66→      const params = new URLSearchParams({ page: String(batchPage), limit: "15", view: "batches" });
    67→      if (batchFilter !== "all") params.set("filter", batchFilter);
    68→      if (batchBranch !== "all") params.set("branchId", batchBranch);
    69→      const data = await api(`/stock?${params}`);
    70→      setBatches(data.items || []);
    71→      setBatchTotal(data.total || 0);
    72→    } catch (e) {
    73→      toast.error(e instanceof Error ? e.message : t("app.error"));
    74→    }
    75→     
    76→  }, [batchPage, batchFilter, batchBranch]);
    77→
    78→  useEffect(() => { if (tab === "movements") loadMovements(); }, [tab, loadMovements]);
    79→  useEffect(() => { if (tab === "expiry") loadBatches(); }, [tab, loadBatches]);
    80→  useEffect(() => { if (tab === "count") loadBatches(); }, [tab, loadBatches]);
    81→
    82→  // ─── الاستلام ───
    83→  const [receiveOpen, setReceiveOpen] = useState(false);
    84→  const [products, setProducts] = useState<any[]>([]);
    85→  const [rForm, setRForm] = useState({ productId: "", branchId: "", batchNo: "", expiryDate: "", qty: "", costPrice: "", shelfLocation: "" });
    86→
    87→  // ─── التحويل ───
    88→  const [transferItems, setTransferItems] = useState<Array<{ productId: string; name: string; qty: number; available: number }>>([]);
    89→  const [trFrom, setTrFrom] = useState("");
    90→  const [trTo, setTrTo] = useState("");
    91→  const [trProduct, setTrProduct] = useState("");
    92→  const [trQty, setTrQty] = useState("");
    93→  const [trAvail, setTrAvail] = useState(0);
    94→
    95→  // ─── التسوية ───
    96→  const [adjustBatch, setAdjustBatch] = useState<any>(null);
    97→  const [adjustQty, setAdjustQty] = useState("");
    98→  const [adjustNote, setAdjustNote] = useState("");
    99→
   100→  // ─── الجرد ───
   101→  const [countBranch, setCountBranch] = useState("");
   102→  const [countItems, setCountItems] = useState<Array<{ batchId: string; name: string; batchNo: string; systemQty: number; countedQty: number }>>([]);
   103→  const [countNote, setCountNote] = useState("");
   104→
   105→  useEffect(() => {
   106→    api("/products?limit=100").then((d) => setProducts(d.items || [])).catch(() => {});
   107→  }, []);
   108→
   109→  const receiveSubmit = async () => {
   110→    try {
   111→      await api("/stock", {
   112→        method: "POST",
   113→        body: {
   114→          action: "receive",
   115→          productId: rForm.productId,
   116→          branchId: rForm.branchId,
   117→          batchNo: rForm.batchNo,
   118→          expiryDate: rForm.expiryDate,
   119→          qty: Number(rForm.qty),
   120→          costPrice: Number(rForm.costPrice),
   121→          shelfLocation: rForm.shelfLocation,
   122→        },
   123→      });
   124→      toast.success(t("inv.done") + " ✓");
   125→      setReceiveOpen(false);
   126→      setRForm({ productId: "", branchId: "", batchNo: "", expiryDate: "", qty: "", costPrice: "", shelfLocation: "" });
   127→      loadBatches();
   128→    } catch (e) {
   129→      toast.error(e instanceof Error ? e.message : t("app.error"));
   130→    }
   131→  };
   132→
   133→  const loadTransferAvail = async (productId: string) => {
   134→    if (!productId || !trFrom) return;
   135→    try {
   136→      const data = await api(`/stock?view=batches&productId=${productId}&branchId=${trFrom}&limit=100`);
   137→      const avail = (data.items || []).filter((b: any) => b.expiryDate > new Date().toISOString()).reduce((s: number, b: any) => s + b.quantity, 0);
   138→      setTrAvail(avail);
   139→    } catch {}
   140→  };
   141→
   142→  const addTransferItem = () => {
   143→    const p = products.find((x) => x.id === trProduct);
   144→    if (!p || !trQty || Number(trQty) <= 0 || !trFrom || !trTo || trFrom === trTo) {
   145→      toast.error(lang === "AR" ? "حدد الفرعين والمنتج والكمية" : "Select branches, product and qty");
   146→      return;
   147→    }
   148→    if (Number(trQty) > trAvail) {
   149→      toast.error(`${t("pos.qtyExceeds")} (${fmtQ(trAvail)})`);
   150→      return;
   151→    }
   152→    if (transferItems.some((x) => x.productId === trProduct)) {
   153→      toast.error(lang === "AR" ? "الصنف مضاف بالفعل" : "Item already added");
   154→      return;
   155→    }
   156→    setTransferItems([...transferItems, { productId: trProduct, name: p.tradeName, qty: Number(trQty), available: trAvail }]);
   157→    setTrProduct("");
   158→    setTrQty("");
   159→    setTrAvail(0);
   160→  };
   161→
   162→  const submitTransfer = async () => {
   163→    if (!transferItems.length) return;
   164→    try {
   165→      await api("/stock", { method: "POST", body: { action: "transfer", fromBranchId: trFrom, toBranchId: trTo, items: transferItems } });
   166→      toast.success(t("inv.done") + " ✓");
   167→      setTransferItems([]);
   168→      loadMovements();
   169→    } catch (e) {
   170→      toast.error(e instanceof Error ? e.message : t("app.error"));
   171→    }
   172→  };
   173→
   174→  const submitAdjust = async () => {
   175→    if (!adjustBatch || adjustQty === "") return;
   176→    try {
   177→      const res = await api("/stock", { method: "POST", body: { action: "adjust", batchId: adjustBatch.id, newQty: Number(adjustQty), note: adjustNote } });
   178→      toast.success(`${t("inv.done")} (${t("inv.variance")}: ${res.variance > 0 ? "+" : ""}${fmtQ(res.variance)})`);
   179→      setAdjustBatch(null);
   180→      loadBatches();
   181→      loadMovements();
   182→    } catch (e) {
   183→      toast.error(e instanceof Error ? e.message : t("app.error"));
   184→    }
   185→  };
   186→
   187→  const startCount = async () => {
   188→    if (!countBranch) {
   189→      toast.error(lang === "AR" ? "حدد الفرع" : "Select branch");
   190→      return;
   191→    }
   192→    try {
   193→      const data = await api(`/stock?view=batches&branchId=${countBranch}&limit=100`);
   194→      const all = data.items || [];
   195→      setCountItems(all.map((b: any) => ({ batchId: b.id, name: b.product.tradeName, batchNo: b.batchNo, systemQty: b.quantity, countedQty: b.quantity })));
   196→      if (!all.length) toast.info(t("app.noData"));
   197→    } catch (e) {
   198→      toast.error(e instanceof Error ? e.message : t("app.error"));
   199→    }
   200→  };
   201→
   202→  const submitCount = async () => {
   203→    if (!countItems.length) return;
   204→    const changed = countItems.filter((i) => i.countedQty !== i.systemQty);
   205→    if (!changed.length) {
   206→      toast.info(lang === "AR" ? "لا فروقات — تم تأكيد الجرد بدون تسويات" : "No variances");
   207→      return;
   208→    }
   209→    try {
   210→      const res = await api("/stock", {
   211→        method: "POST",
   212→        body: { action: "stockCount", branchId: countBranch, items: changed.map((i) => ({ batchId: i.batchId, countedQty: i.countedQty })), note: countNote },
   213→      });
   214→      toast.success(`${t("inv.done")} — ${lang === "AR" ? "فروقات" : "variances"}: ${fmtQ(res.totalVariance)}`);
   215→      setCountItems([]);
   216→      loadBatches();
   217→      loadMovements();
   218→    } catch (e) {
   219→      toast.error(e instanceof Error ? e.message : t("app.error"));
   220→    }
   221→  };
   222→
   223→  const expBadge = (d: string) => {
   224→    const st = expiryStatus(d);
   225→    const days = Math.ceil((new Date(d).getTime() - Date.now()) / 86400000);
   226→    return (
   227→      <Badge
   228→        variant={st === "expired" ? "destructive" : st === "week" ? "destructive" : st === "month" ? "secondary" : "outline"}
   229→        className={cn(st === "week" && "bg-red-100 text-red-700 border-0", st === "month" && "bg-amber-100 text-amber-700 border-0")}
   230→      >
   231→        {st === "expired" ? t("prod.expiredBatch") : `${fmtQ(days)} ${t("prod.days")}`}
   232→      </Badge>
   233→    );
   234→  };
   235→
   236→  return (
   237→    <div>
   238→      <PageHeader
   239→        icon={Warehouse}
   240→        title={t("inv.title")}
   241→        actions={
   242→          <Button onClick={() => setReceiveOpen(true)} className="gap-2">
   243→            <PackagePlus className="h-4 w-4" /> {t("inv.receive")}
   244→          </Button>
   245→        }
   246→      />
   247→
   248→      <Tabs value={tab} onValueChange={setTab} dir={lang === "AR" ? "rtl" : "ltr"}>
   249→        <TabsList className="flex-wrap h-auto">
   250→          <TabsTrigger value="movements" className="gap-1.5"><RefreshCw className="h-4 w-4" /> {t("inv.movements")}</TabsTrigger>
   251→          <TabsTrigger value="expiry" className="gap-1.5"><CalendarClock className="h-4 w-4" /> {t("inv.expiry")}</TabsTrigger>
   252→          <TabsTrigger value="transfer" className="gap-1.5"><ArrowLeftRight className="h-4 w-4" /> {t("inv.transfer")}</TabsTrigger>
   253→          <TabsTrigger value="adjust" className="gap-1.5"><Scale className="h-4 w-4" /> {t("inv.adjust")}</TabsTrigger>
   254→          <TabsTrigger value="count" className="gap-1.5"><ClipboardCheck className="h-4 w-4" /> {t("inv.count")}</TabsTrigger>
   255→        </TabsList>
   256→
   257→        {/* ─── الحركات ─── */}
   258→        <TabsContent value="movements">
   259→          <Card>
   260→            <CardContent className="p-4">
   261→              <div className="flex flex-wrap gap-2 mb-3">
   262→                <Select value={movType} onValueChange={(v) => { setMovType(v); setMovPage(1); }}>
   263→                  <SelectTrigger className="w-40"><SelectValue /></SelectTrigger>
   264→                  <SelectContent>
   265→                    <SelectItem value="all">{t("app.all")}</SelectItem>
   266→                    {["RECEIVE", "SALE", "SALE_REFUND", "TRANSFER_OUT", "TRANSFER_IN", "ADJUSTMENT", "STOCK_COUNT", "ISSUE"].map((m) => (
   267→                      <SelectItem key={m} value={m}>{t(`inv.type.${m}`)}</SelectItem>
   268→                    ))}
   269→                  </SelectContent>
   270→                </Select>
   271→                <Select value={movBranch} onValueChange={(v) => { setMovBranch(v); setMovPage(1); }}>
   272→                  <SelectTrigger className="w-44"><SelectValue /></SelectTrigger>
   273→                  <SelectContent>
   274→                    <SelectItem value="all">{t("app.all")}</SelectItem>
   275→                    {branches.map((b) => <SelectItem key={b.id} value={b.id}>{lang === "AR" ? b.name : b.nameEn}</SelectItem>)}
   276→                  </SelectContent>
   277→                </Select>
   278→              </div>
   279→              {movements.length === 0 ? (
   280→                <EmptyState message={t("app.noData")} />
   281→              ) : (
   282→                <div className="overflow-x-auto">
   283→                  <Table>
   284→                    <TableHeader>
   285→                      <TableRow>
   286→                        <TableHead>{t("app.date")}</TableHead>
   287→                        <TableHead>{t("prod.tradeName")}</TableHead>
   288→                        <TableHead>{t("app.type")}</TableHead>
   289→                        <TableHead className="text-center">{t("app.quantity")}</TableHead>
   290→                        <TableHead>{t("app.branch")}</TableHead>
   291→                        <TableHead>{lang === "AR" ? "المستخدم" : "User"}</TableHead>
   292→                      </TableRow>
   293→                    </TableHeader>
   294→                    <TableBody>
   295→                      {movements.map((m) => (
   296→                        <TableRow key={m.id}>
   297→                          <TableCell className="text-xs whitespace-nowrap">{fmtDate(m.createdAt, lang, true)}</TableCell>
   298→                          <TableCell className="text-sm font-medium">{m.product?.tradeName || "-"}</TableCell>
   299→                          <TableCell><Badge variant="outline">{t(`inv.type.${m.type}`)}</Badge></TableCell>
   300→                          <TableCell className={cn("text-center font-bold", m.quantity < 0 ? "text-red-500" : "text-emerald-600")}>{m.quantity > 0 ? `+${fmtQ(m.quantity)}` : fmtQ(m.quantity)}</TableCell>
   301→                          <TableCell className="text-xs">{lang === "AR" ? m.branch?.name : m.branch?.nameEn}</TableCell>
   302→                          <TableCell className="text-xs">{m.user?.fullName || "-"}</TableCell>
   303→                        </TableRow>
   304→                      ))}
   305→                    </TableBody>
   306→                  </Table>
   307→                </div>
   308→              )}
   309→              <Pagination page={movPage} pageSize={15} total={movTotal} onPage={setMovPage} />
   310→            </CardContent>
   311→          </Card>
   312→        </TabsContent>
   313→
   314→        {/* ─── الصلاحية والتشغيلات ─── */}
   315→        <TabsContent value="expiry">
   316→          <Card>
   317→            <CardContent className="p-4">
   318→              <div className="flex flex-wrap gap-2 mb-3">
   319→                <Select value={batchFilter} onValueChange={(v) => { setBatchFilter(v); setBatchPage(1); }}>
   320→                  <SelectTrigger className="w-44"><SelectValue /></SelectTrigger>
   321→                  <SelectContent>
   322→                    <SelectItem value="all">{t("app.all")}</SelectItem>
   323→                    <SelectItem value="EXPIRING">{lang === "AR" ? "أقل من 90 يوم" : "Within 90 days"}</SelectItem>
   324→                    <SelectItem value="EXPIRED">{t("dash.expired")}</SelectItem>
   325→                  </SelectContent>
   326→                </Select>
   327→                <Select value={batchBranch} onValueChange={(v) => { setBatchBranch(v); setBatchPage(1); }}>
   328→                  <SelectTrigger className="w-44"><SelectValue /></SelectTrigger>
   329→                  <SelectContent>
   330→                    <SelectItem value="all">{t("app.all")}</SelectItem>
   331→                    {branches.map((b) => <SelectItem key={b.id} value={b.id}>{lang === "AR" ? b.name : b.nameEn}</SelectItem>)}
   332→                  </SelectContent>
   333→                </Select>
   334→              </div>
   335→              {batches.length === 0 ? (
   336→                <EmptyState message={t("app.noData")} />
   337→              ) : (
   338→                <div className="overflow-x-auto">
   339→                  <Table>
   340→                    <TableHeader>
   341→                      <TableRow>
   342→                        <TableHead>{t("prod.tradeName")}</TableHead>
   343→                        <TableHead>{t("prod.batchNo")}</TableHead>
   344→                        <TableHead>{t("prod.expiry")}</TableHead>
   345→                        <TableHead>{t("inv.days")}</TableHead>
   346→                        <TableHead className="text-center">{t("app.quantity")}</TableHead>
   347→                        <TableHead className="text-end">{t("inv.costPrice")}</TableHead>
   348→                        <TableHead>{t("app.branch")}</TableHead>
   349→                        <TableHead></TableHead>
   350→                      </TableRow>
   351→                    </TableHeader>
   352→                    <TableBody>
   353→                      {batches.map((b) => (
   354→                        <TableRow key={b.id} className={cn(expiryStatus(b.expiryDate) === "expired" && "bg-red-50/50 dark:bg-red-950/20")}>
   355→                          <TableCell className="font-medium text-sm">{b.product.tradeName}</TableCell>
   356→                          <TableCell className="font-mono text-xs" dir="ltr">{b.batchNo}</TableCell>
   357→                          <TableCell className="text-xs whitespace-nowrap">{fmtDate(b.expiryDate, lang)}</TableCell>
   358→                          <TableCell>{expBadge(b.expiryDate)}</TableCell>
   359→                          <TableCell className="text-center font-bold">{fmtQ(b.quantity)}</TableCell>
   360→                          <TableCell className="text-end text-xs">{fmtM(b.costPrice * b.quantity)}</TableCell>
   361→                          <TableCell className="text-xs">{lang === "AR" ? b.branch?.name : b.branch?.nameEn}</TableCell>
   362→                          <TableCell>
   363→                            <Button variant="outline" size="sm" onClick={() => { setAdjustBatch(b); setAdjustQty(String(b.quantity)); setAdjustNote(""); }}>
   364→                              <Scale className="h-3.5 w-3.5" />
   365→                            </Button>
   366→                          </TableCell>
   367→                        </TableRow>
   368→                      ))}
   369→                    </TableBody>
   370→                  </Table>
   371→                </div>
   372→              )}
   373→              <Pagination page={batchPage} pageSize={15} total={batchTotal} onPage={setBatchPage} />
   374→            </CardContent>
   375→          </Card>
   376→        </TabsContent>
   377→
   378→        {/* ─── التحويل ─── */}
   379→        <TabsContent value="transfer">
   380→          <Card>
   381→            <CardHeader><CardTitle className="text-base flex items-center gap-2"><ArrowLeftRight className="h-5 w-5 text-primary" /> {t("inv.transfer")}</CardTitle></CardHeader>
   382→            <CardContent className="space-y-4">
   383→              <div className="grid grid-cols-1 md:grid-cols-2 gap-3">
   384→                <div>
   385→                  <Label>{t("inv.fromBranch")}</Label>
   386→                  <Select value={trFrom} onValueChange={(v) => { setTrFrom(v); setTrAvail(0); }}>
   387→                    <SelectTrigger className="mt-1"><SelectValue placeholder="—" /></SelectTrigger>
   388→                    <SelectContent>
   389→                      {branches.map((b) => <SelectItem key={b.id} value={b.id}>{lang === "AR" ? b.name : b.nameEn}</SelectItem>)}
   390→                    </SelectContent>
   391→                  </Select>
   392→                </div>
   393→                <div>
   394→                  <Label>{t("inv.toBranch")}</Label>
   395→                  <Select value={trTo} onValueChange={setTrTo}>
   396→                    <SelectTrigger className="mt-1"><SelectValue placeholder="—" /></SelectTrigger>
   397→                    <SelectContent>
   398→                      {branches.filter((b) => b.id !== trFrom).map((b) => <SelectItem key={b.id} value={b.id}>{lang === "AR" ? b.name : b.nameEn}</SelectItem>)}
   399→                    </SelectContent>
   400→                  </Select>
   401→                </div>
   402→              </div>
   403→              <div className="grid grid-cols-1 md:grid-cols-12 gap-2 items-end">
   404→                <div className="md:col-span-6">
   405→                  <Label>{t("prod.tradeName")}</Label>
   406→                  <Select value={trProduct} onValueChange={(v) => { setTrProduct(v); loadTransferAvail(v); }}>
   407→                    <SelectTrigger className="mt-1"><SelectValue placeholder="—" /></SelectTrigger>
   408→                    <SelectContent className="max-h-52">
   409→                      {products.map((p) => <SelectItem key={p.id} value={p.id}>{p.tradeName}</SelectItem>)}
   410→                    </SelectContent>
   411→                  </Select>
   412→                </div>
   413→                <div className="md:col-span-2">
   414→                  <Label>{t("app.quantity")}</Label>
   415→                  <Input type="number" min={1} value={trQty} onChange={(e) => setTrQty(e.target.value)} className="mt-1" />
   416→                </div>
   417→                <div className="md:col-span-2 text-xs text-muted-foreground pb-2">
   418→                  {trProduct && <>{t("pos.available")}: <span className="font-bold text-foreground">{fmtQ(trAvail)}</span></>}
   419→                </div>
   420→                <div className="md:col-span-2">
   421→                  <Button onClick={addTransferItem} className="w-full gap-1.5"><Plus className="h-4 w-4" /> {t("app.add")}</Button>
   422→                </div>
   423→              </div>
   424→
   425→              {transferItems.length > 0 && (
   426→                <>
   427→                  <div className="space-y-2">
   428→                    {transferItems.map((it, i) => (
   429→                      <div key={i} className="flex items-center justify-between p-2.5 rounded border text-sm">
   430→                        <span className="font-medium">{it.name}</span>
   431→                        <div className="flex items-center gap-3">

   578→                <SelectContent>
   579→                  {branches.map((b) => <SelectItem key={b.id} value={b.id}>{lang === "AR" ? b.name : b.nameEn}</SelectItem>)}

   583→            <div className="grid grid-cols-2 gap-2">
   584→              <div><Label>{t("prod.batchNo")} *</Label><Input dir="ltr" value={rForm.batchNo} onChange={(e) => setRForm({ ...rForm, batchNo: e.target.value })} className="mt-1 font-mono" /></div>

   589→            </div>

   597→      {/* تسوية */}
   598→      <Dialog open={!!adjustBatch} onOpenChange={(o) => !o && setAdjustBatch(null)}>
   599→        <DialogContent className="max-w-sm">
   600→          <DialogHeader>
   601→            <DialogTitle className="flex items-center gap-2"><Scale className="h-5 w-5 text-primary" /> {t("inv.adjust")}</DialogTitle>
   602→          </DialogHeader>
   603→          {adjustBatch && (
   604→            <div className="space-y-3">
   605→              <div className="p-3 rounded bg-muted/50 text-sm">
   606→                <p className="font-semibold">{adjustBatch.product.tradeName}</p>

   618→              </div>
   619→              <div><Label>{t("app.notes")}</Label><Input value={adjustNote} onChange={(e) => setAdjustNote(e.target.value)} className="mt-1" /></div>
   620→              <Button onClick={submitAdjust} className="w-full">{t("app.save")}</Button>
